Flight::route('GET /talleres/@lang', function ($lang) {
  $items = [];
  // Solo talleres activos. Los ya pasados quedan al final (fecha NULL al ultimo).
  $sql = 'SELECT * FROM talleres WHERE active = 1 ORDER BY (fecha IS NULL), (fecha < CURDATE()), fecha ASC, top DESC';
  $statement = Flight::db()->prepare($sql);
  $statement->execute();
  $results = $statement->fetchAll();

  foreach ($results as $row) {
    $es = $lang == 'es';
    $items[] = [
      'id' => intval($row['id']),
      'slug' => $row['slug'],
      'title' => $es ? $row['title'] : ($row['title_en'] ?: $row['title']),
      'text' => $es ? $row['text'] : ($row['text_en'] ?: $row['text']),
      'fecha' => $row['fecha'],
      'horario' => $row['horario'],
      'cupos' => intval($row['cupos']),
      'price_ars' => $row['price_ars'],
      'price_usd' => $row['price_usd'],
      'image' => $row['image'] ? IMAGESURL . $row['image'] : null,
      'form_url' => $row['form_url'] ?: null,
    ];
  }
  Flight::json($items ?: null);
});

Flight::route('GET /taller/@slug/@lang', function ($slug, $lang) {
  $statement = Flight::db()->prepare('SELECT * FROM talleres WHERE slug = ? AND active = 1');
  $statement->execute([$slug]);
  $row = $statement->fetch();
  if (!$row) {
    Flight::json(null);
    return;
  }
  $es = $lang == 'es';
  Flight::json([
    'id' => intval($row['id']),
    'slug' => $row['slug'],
    'title' => $es ? $row['title'] : ($row['title_en'] ?: $row['title']),
    'text' => $es ? $row['text'] : ($row['text_en'] ?: $row['text']),
    'fecha' => $row['fecha'],
    'horario' => $row['horario'],
    'cupos' => intval($row['cupos']),
    'price_ars' => $row['price_ars'],
    'price_usd' => $row['price_usd'],
    'image' => $row['image'] ? IMAGESURL . $row['image'] : null,
    'form_url' => $row['form_url'] ?: null,
  ]);
});
